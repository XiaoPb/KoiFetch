# Backend Python dependencies for the Koi Fetch v1 foundation.

# power the ASGI application entrypoint (app/main.py) and the local dev server;
# httpx backs FastAPI's TestClient. Later tasks will extend this file as they
# introduce their own dependencies.

# Test dependency: backend/tests/test_compose.py parses docker-compose.yml with
# PyYAML (Docker itself is absent in CI, so the compose contract is validated
# statically). Declared explicitly so a fresh install never silently skips the
# compose suite because PyYAML arrived only as a transitive dependency.
PyYAML>=6.0

# Test dependency: backend/tests/test_download_ws.py drives the progress
# WebSocket with websockets.sync.client (added in websockets 12.0). Not a
# runtime dependency — the app's own WS server runs through Starlette.
websockets>=12.0

# Persistence: SQLAlchemy ORM (app/infrastructure/models.py), Alembic
# migrations (backend/alembic, run via `alembic upgrade head` at deploy time),
# and bcrypt for the admin password hash (app/infrastructure/seed.py).
# bcrypt is pinned below 5; across the allowed range (>=3.2,<5) bcrypt SILENTLY
# TRUNCATES >72-byte passwords (no exception) and the hash format is identical.
# The seed and auth service therefore reject >72-byte inputs explicitly
# (app/infrastructure/seed.py, app/application/auth_service.py).
sqlalchemy>=2.0
alembic>=1.13
bcrypt>=3.2,<5

# Tokens: PyJWT signs configurable admin access tokens (seven-day default)
# and five-minute reusable file tokens issued by the download API; each token
# binds a download task+filename and playback may send repeated GET/Range
# requests (app/adapters/tokens_jwt.py).
pyjwt>=2.13

# Scheduling: APScheduler runs the worker daemon's periodic bubble-cleanup /
# stale-task-expiry job (app/workers/cleanup.py). Pinned below 4 so the
# 3.x BackgroundScheduler API stays stable.
